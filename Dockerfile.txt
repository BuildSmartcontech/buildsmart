# Dockerfile para Hugging Face Spaces
FROM python:3.11-slim

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

EXPOSE 7860

CMD ["sh", "-c", "uvicorn backend.main:app --host 0.0.0.0 --port 7860 & streamlit run app.py --server.port 7861 --server.enableCORS false"]
